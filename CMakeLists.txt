cmake_minimum_required(VERSION 3.20)
project(TinyFault VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(tinyfault STATIC src/tinyfault.c)
target_include_directories(tinyfault PUBLIC src)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(tinyfault PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(tinyfault_tests tests/test_tinyfault.c)
  target_link_libraries(tinyfault_tests PRIVATE tinyfault)
  if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(tinyfault_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
  endif()
  add_test(NAME tinyfault_tests COMMAND tinyfault_tests)
  add_executable(tinyfault_reset_tests src/tf_esp32_reset.c tests/test_esp32_reset.c)
  target_include_directories(tinyfault_reset_tests PRIVATE src tests/fakes)
  target_compile_definitions(tinyfault_reset_tests PRIVATE ESP_PLATFORM)
  if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(tinyfault_reset_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
  endif()
  add_test(NAME tinyfault_reset_tests COMMAND tinyfault_reset_tests)
  foreach(mode enabled disabled)
    add_executable(tinyfault_coredump_${mode}_tests src/tf_esp32_coredump.c
                   tests/test_esp32_coredump.c)
    target_include_directories(tinyfault_coredump_${mode}_tests PRIVATE src tests/fakes)
    target_compile_definitions(tinyfault_coredump_${mode}_tests PRIVATE ESP_PLATFORM)
    if(mode STREQUAL "enabled")
      target_compile_definitions(tinyfault_coredump_${mode}_tests PRIVATE
                                 CONFIG_ESP_COREDUMP_ENABLE_TO_FLASH=1
                                 CONFIG_ESP_COREDUMP_DATA_FORMAT_ELF=1)
    else()
      target_compile_definitions(tinyfault_coredump_${mode}_tests PRIVATE TEST_DISABLED)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
      target_compile_options(tinyfault_coredump_${mode}_tests PRIVATE
                             -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
    endif()
    add_test(NAME tinyfault_coredump_${mode}_tests COMMAND tinyfault_coredump_${mode}_tests)
  endforeach()
  add_executable(tinyfault_avr_reset_tests tests/test_avr_reset.c)
  target_include_directories(tinyfault_avr_reset_tests PRIVATE src)
  if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(tinyfault_avr_reset_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
  endif()
  add_test(NAME tinyfault_avr_reset_tests COMMAND tinyfault_avr_reset_tests)
  add_executable(tinyfault_nvs_tests tests/test_nvs.cpp)
  target_link_libraries(tinyfault_nvs_tests PRIVATE tinyfault)
  target_include_directories(tinyfault_nvs_tests PRIVATE tests/fakes)
  target_compile_definitions(tinyfault_nvs_tests PRIVATE ARDUINO_ARCH_ESP32)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(tinyfault_nvs_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
  endif()
  add_test(NAME tinyfault_nvs_tests COMMAND tinyfault_nvs_tests)
  add_executable(tinyfault_eeprom_tests tests/test_eeprom.cpp)
  target_link_libraries(tinyfault_eeprom_tests PRIVATE tinyfault)
  target_include_directories(tinyfault_eeprom_tests PRIVATE tests/fakes)
  target_compile_definitions(tinyfault_eeprom_tests PRIVATE ARDUINO_ARCH_AVR)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(tinyfault_eeprom_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
  endif()
  add_test(NAME tinyfault_eeprom_tests COMMAND tinyfault_eeprom_tests)
  add_executable(tinyfault_export_tests tests/test_export.cpp)
  target_link_libraries(tinyfault_export_tests PRIVATE tinyfault)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(tinyfault_export_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
  endif()
  add_test(NAME tinyfault_export_tests COMMAND tinyfault_export_tests)
  add_executable(tinyfault_incident_tests tests/test_incident.cpp)
  target_link_libraries(tinyfault_incident_tests PRIVATE tinyfault)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(tinyfault_incident_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
  endif()
  add_test(NAME tinyfault_incident_tests COMMAND tinyfault_incident_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME tinyfault_decode_tests COMMAND ${Python3_EXECUTABLE}
           ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_decode.py $<TARGET_FILE:tinyfault_export_tests>)
  set_tests_properties(tinyfault_decode_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME tinyfault_correlate_tests COMMAND ${Python3_EXECUTABLE}
           ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_correlate.py)
  set_tests_properties(tinyfault_correlate_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME tinyfault_hardware_avr_tests COMMAND ${Python3_EXECUTABLE}
           ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_hardware_avr.py)
  set_tests_properties(tinyfault_hardware_avr_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME tinyfault_hardware_incident_tests COMMAND ${Python3_EXECUTABLE}
           ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_hardware_incident.py)
  set_tests_properties(tinyfault_hardware_incident_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME tinyfault_hardware_avr_incident_tests COMMAND ${Python3_EXECUTABLE}
           ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_hardware_avr_incident.py)
  set_tests_properties(tinyfault_hardware_avr_incident_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endif()
