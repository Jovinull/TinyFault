cmake_minimum_required(VERSION 3.20)
project(TinyFault VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(tinyfault STATIC src/tinyfault.c)
target_include_directories(tinyfault PUBLIC src)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(tinyfault PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(tinyfault_tests tests/test_tinyfault.c)
  target_link_libraries(tinyfault_tests PRIVATE tinyfault)
  if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(tinyfault_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
  endif()
  add_test(NAME tinyfault_tests COMMAND tinyfault_tests)
  add_executable(tinyfault_reset_tests src/tf_esp32_reset.c tests/test_esp32_reset.c)
  target_include_directories(tinyfault_reset_tests PRIVATE src tests/fakes)
  target_compile_definitions(tinyfault_reset_tests PRIVATE ESP_PLATFORM)
  if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(tinyfault_reset_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
  endif()
  add_test(NAME tinyfault_reset_tests COMMAND tinyfault_reset_tests)
  add_executable(tinyfault_nvs_tests tests/test_nvs.cpp)
  target_link_libraries(tinyfault_nvs_tests PRIVATE tinyfault)
  target_include_directories(tinyfault_nvs_tests PRIVATE tests/fakes)
  target_compile_definitions(tinyfault_nvs_tests PRIVATE ARDUINO_ARCH_ESP32)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(tinyfault_nvs_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
  endif()
  add_test(NAME tinyfault_nvs_tests COMMAND tinyfault_nvs_tests)
endif()
